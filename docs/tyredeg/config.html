tyredeg.config

Load and validate the project configuration (config.yaml).

The rest of the code never reads the YAML file directly. It calls load_config(), which returns a Config object whose values have already been checked for type and range. This means a typo in config.yaml fails immediately with a clear message, instead of causing a confusing error deep inside a numerical method later on.

Structure of this module
  1. Small _require_* helpers, each checking one kind of value.
  2. Dataclasses that hold the validated values (RaceSpec, Config).
  3. config_from_dict(), which validates a plain dictionary section by section.
  4. load_config(), which reads the YAML file and calls config_from_dict.

 52def _require_int(value, name, minimum=None, maximum=None):
 53    """Return ``value`` if it is an int within [minimum, maximum], else raise.
 54
 55    ``bool`` is rejected explicitly because in Python ``True`` is an ``int``
 56    (``True == 1``), so ``random_seed: true`` would otherwise be accepted.
 57    """
 58    # Reject booleans explicitly: in Python True is also an int.
 59    if isinstance(value, bool) or not isinstance(value, int):
 60        raise TypeError(f"'{name}' must be an integer, got {value!r}.")
 61    # Enforce the optional lower and upper bounds.
 62    if minimum is not None and value < minimum:
 63        raise ValueError(f"'{name}' must be >= {minimum}, got {value}.")
 64    if maximum is not None and value > maximum:
 65        raise ValueError(f"'{name}' must be <= {maximum}, got {value}.")
 66    return value
 67

 69def _require_float(value, name, minimum=None, strictly_positive=False):
 70    """Return ``value`` as a float if it is a real number in range, else raise.
 71
 72    Integers are accepted (YAML reads ``90`` as an int) and converted to float.
 73    """
 74    # Accept ints and floats (YAML may give either), but not booleans or strings.
 75    if isinstance(value, bool) or not isinstance(value, (int, float)):
 76        raise TypeError(f"'{name}' must be a number, got {value!r}.")
 77    value = float(value)
 78    # NaN compares False with everything, so it would slip through the range
 79    # checks below; reject it explicitly.
 80    if value != value:
 81        raise ValueError(f"'{name}' must not be NaN.")
 82    # Enforce the optional positivity and lower bound.
 83    if strictly_positive and value <= 0:
 84        raise ValueError(f"'{name}' must be > 0, got {value}.")
 85    if minimum is not None and value < minimum:
 86        raise ValueError(f"'{name}' must be >= {minimum}, got {value}.")
 87    return value

 90def _require_string(value, name):
 91    """Return ``value`` stripped of surrounding spaces if it is a non-empty string."""
 92    # A string that is not empty after removing surrounding spaces.
 93    if not isinstance(value, str):
 94        raise TypeError(f"'{name}' must be a string, got {value!r}.")
 95    if value.strip() == "":
 96        raise ValueError(f"'{name}' must not be empty.")
 97    return value.strip()

100def _require_string_list(value, name, allowed=None):
101    """Return a tuple of strings, or ``None`` if ``value`` is ``None``.
102
103    ``None`` means "no filter" (keep everything). An empty list is rejected
104    because it would silently select nothing.
105    """
106    # null means "no filter" (e.g. all teams), so it is passed through as None.
107    if value is None:
108        return None
109    # Otherwise it must be a non-empty list of allowed strings.
110    if not isinstance(value, list):
111        raise TypeError(f"'{name}' must be a list of strings or null, got {value!r}.")
112    if len(value) == 0:
113        raise ValueError(f"'{name}' must not be an empty list; use null for 'all'.")
114    items = []
115    for item in value:
116        item = _require_string(item, f"{name} entry")
117        if allowed is not None and item not in allowed:
118            raise ValueError(
119                f"Unknown value {item!r} in '{name}'. Allowed values: {allowed}."
120            )
121        items.append(item)
122    return tuple(items)

125def _require_section(raw, name):
126    """Return the sub-dictionary ``raw[name]``, raising if it is missing or not a mapping."""
127    # A top-level section such as 'fuel:' must exist and be a mapping.
128    if name not in raw:
129        raise ValueError(f"Missing section '{name}' in the configuration.")
130    section = raw[name]
131    if not isinstance(section, dict):
132        raise TypeError(f"Section '{name}' must be a mapping, got {section!r}.")
133    return section

136def _require_key(section, key, section_name):
137    """Return ``section[key]``, raising a clear error if the key is absent.
138
139    The key must be present even when its value is ``null``, so that a
140    misspelt key name is caught rather than silently treated as "not set".
141    """
142    # A missing key is reported with the section it belongs to.
143    if key not in section:
144        raise ValueError(f"Missing key '{key}' in section '{section_name}'.")
145    return section[key]

421def _parse_race(entry, index, last_year):
422    """Validate one entry of ``data.races`` and return a :class:`RaceSpec`."""
423    # Name used in error messages, e.g. "data.races[2]".
424    name = f"data.races[{index}]"
425    if not isinstance(entry, dict):
426        raise TypeError(f"'{name}' must be a mapping with year/event/role, got {entry!r}.")
427    # Each race needs a valid year, an event name, a train/test role and a data source.
428    year = _require_int(
429        _require_key(entry, "year", name), f"{name}.year", FIRST_SUPPORTED_YEAR, last_year
430    )
431    event = _require_string(_require_key(entry, "event", name), f"{name}.event")
432    role = _require_string(_require_key(entry, "role", name), f"{name}.role")
433    if role not in VALID_RACE_ROLES:
434        raise ValueError(f"'{name}.role' must be one of {VALID_RACE_ROLES}, got {role!r}.")
435    source = _require_string(_require_key(entry, "source", name), f"{name}.source")
436    if source not in VALID_SOURCES:
437        raise ValueError(f"'{name}.source' must be one of {VALID_SOURCES}, got {source!r}.")
438    # OpenF1 has no free data before 2023, so catch that combination here.
439    if source == "openf1" and year < OPENF1_FIRST_YEAR:
440        raise ValueError(
441            f"'{name}': OpenF1 only has free data from {OPENF1_FIRST_YEAR} onward, got {year}."
442        )
443    return RaceSpec(year=year, event=event, role=role, source=source)

446def _parse_races(raw_races):
447    """Validate the whole ``data.races`` list.
448
449    Checks that there is at least one training and one test race and that no
450    race is listed twice. A duplicated race could end up in both the training
451    and the test set, which would leak test information into model selection.
452    """
453    # The race list must not be empty.
454    if not isinstance(raw_races, list) or len(raw_races) == 0:
455        raise ValueError("'data.races' must be a non-empty list of races.")
456
457    # A season cannot be later than the current calendar year.
458    # Validate every race entry; a season cannot be later than the current year.
459    last_year = datetime.date.today().year
460    races = []
461    for index, entry in enumerate(raw_races):
462        races.append(_parse_race(entry, index, last_year))
463
464    # The same race listed twice could end up in both training and test data (leakage).
465    seen = set()
466    for race in races:
467        key = (race.year, race.event.casefold())
468        if key in seen:
469            raise ValueError(f"Race {race.year} {race.event!r} is listed more than once.")
470        seen.add(key)
471
472    # Both a training race and a test race are required.
473    roles = {race.role for race in races}
474    if "train" not in roles:
475        raise ValueError("At least one race must have role 'train'.")
476    if "test" not in roles:
477        raise ValueError("At least one race must have role 'test'.")
478    return tuple(races)

481def _optional(value, check, *args, **kwargs):
482    """Return ``None`` if ``value`` is ``None``, otherwise ``check(value, ...)``.
483
484    Used for keys that are still ``null`` because they are decided in a later
485    phase: the range is only checked once a value has been filled in.
486    """
487    # Keys set to null are decided later: only check a value that is actually given.
488    if value is None:
489        return None
490    return check(value, *args, **kwargs)

493def _parse_k_fuel_range(fuel, k_fuel):
494    """Validate ``fuel.k_fuel_min`` / ``fuel.k_fuel_max`` (sensitivity range).
495
496    Returns ``None`` if both are null, otherwise ``(min, max)``. The default
497    ``k_fuel`` must lie inside the range, otherwise the sensitivity analysis
498    would not bracket the value actually used.
499    """
500    # The k_fuel range for the sensitivity analysis: both ends given or both null.
501    low = _optional(_require_key(fuel, "k_fuel_min", "fuel"), _require_float, "fuel.k_fuel_min", 0.0)
502    high = _optional(_require_key(fuel, "k_fuel_max", "fuel"), _require_float, "fuel.k_fuel_max", 0.0)
503    if low is None and high is None:
504        return None
505    if low is None or high is None:
506        raise ValueError("'fuel.k_fuel_min' and 'fuel.k_fuel_max' must both be set or both be null.")
507    if low > high:
508        raise ValueError(f"'fuel.k_fuel_min' ({low}) must be <= 'fuel.k_fuel_max' ({high}).")
509    # The nominal k_fuel must lie inside its own sensitivity range.
510    if k_fuel is not None and not low <= k_fuel <= high:
511        raise ValueError(f"'fuel.k_fuel' ({k_fuel}) must lie within [{low}, {high}].")
512    return (low, high)

523def _parse_ode(section):
524    """Validate ``numerics.ode``."""
525    # Read every ODE setting as a positive number (max_steps as an integer >= 1).
526    name = "numerics.ode"
527    settings = OdeSettings(
528        rtol=_positive_float(section, "rtol", name),
529        atol=_positive_float(section, "atol", name),
530        initial_step_fraction=_positive_float(section, "initial_step_fraction", name),
531        max_steps=_require_int(_require_key(section, "max_steps", name), f"{name}.max_steps", 1),
532        safety=_positive_float(section, "safety", name),
533        min_factor=_positive_float(section, "min_factor", name),
534        max_factor=_positive_float(section, "max_factor", name),
535    )
536    # A safety factor above 1 would aim for steps that are expected to fail.
537    # The safety factor only shrinks steps, and the step must be able to shrink and grow.
538    if settings.safety > 1:
539        raise ValueError(f"'{name}.safety' must be <= 1, got {settings.safety}.")
540    # The controller must be able to shrink (min < 1) and to grow (max > 1).
541    if not settings.min_factor < 1 < settings.max_factor:

617def _parse_models(raw):
618    """Validate the ``models`` section (Phase 4)."""
619    # Read the three model sections.
620    models = _require_section(raw, "models")
621    poly = _require_section(models, "polynomial")
622    ode = _require_section(models, "wear_ode")
623    knn = _require_section(models, "knn")
624
625    # Polynomial: candidate degrees and a default that is one of them.
626    degrees = _int_list(_require_key(poly, "candidate_degrees", "models.polynomial"),
627                        "models.polynomial.candidate_degrees", 1)
628    default_degree = _require_int(_require_key(poly, "default_degree", "models.polynomial"),
629                                  "models.polynomial.default_degree", 1)
630    if default_degree not in degrees:
631        raise ValueError("'models.polynomial.default_degree' must be one of the candidate degrees.")
632
633    # Wear ODE: the search bracket for b must be [low, high] with low < high.
634    bracket = _require_key(ode, "b_bracket", "models.wear_ode")
635    if not isinstance(bracket, list) or len(bracket) != 2:
636        raise TypeError("'models.wear_ode.b_bracket' must be a [low, high] list.")
637    low = _require_float(bracket[0], "models.wear_ode.b_bracket[0]")
638    high = _require_float(bracket[1], "models.wear_ode.b_bracket[1]")
639    if not low < high:
640        raise ValueError("'models.wear_ode.b_bracket' must satisfy low < high.")
641    # The ODE solver must be a known one, and at least one search tolerance must be positive.
642    solver = _require_string(_require_key(ode, "solver", "models.wear_ode"), "models.wear_ode.solver")
643    if solver not in VALID_ODE_SOLVERS:
644        raise ValueError(f"'models.wear_ode.solver' must be one of {VALID_ODE_SOLVERS}, got {solver!r}.")
645    b_xtol = _require_float(_require_key(ode, "b_xtol", "models.wear_ode"), "models.wear_ode.b_xtol", 0.0)
646    b_ftol = _require_float(_require_key(ode, "b_ftol", "models.wear_ode"), "models.wear_ode.b_ftol", 0.0)
647    if b_xtol == 0 and b_ftol == 0:
648        raise ValueError("'models.wear_ode': b_xtol and b_ftol cannot both be 0.")
649
650    # KNN: candidate k values, weightings, and a default from among them.
651    candidate_k = _int_list(_require_key(knn, "candidate_k", "models.knn"), "models.knn.candidate_k", 1)
652    weights = _require_string_list(_require_key(knn, "candidate_weights", "models.knn"),
653                                   "models.knn.candidate_weights", VALID_KNN_WEIGHTS)
654    if weights is None:
655        raise ValueError("'models.knn.candidate_weights' must list at least one weighting.")
656    default_k = _require_int(_require_key(knn, "default_k", "models.knn"), "models.knn.default_k", 1)
657    default_weights = _require_string(_require_key(knn, "default_weights", "models.knn"),
658                                      "models.knn.default_weights")
659    if default_weights not in weights:
660        raise ValueError("'models.knn.default_weights' must be one of the candidate weightings.")
661
662    # Confidence level for bootstrap intervals must lie in (0, 1).
663    confidence = _positive_float(models, "confidence_level", "models")
664    if confidence >= 1:
665        raise ValueError(f"'models.confidence_level' must be < 1, got {confidence}.")
666
667    # Assemble all model settings into one validated object.
668    return ModelSettings(
669        candidate_degrees=degrees,
670        default_degree=default_degree,
671        b_bracket=(low, high),
672        b_grid_points=_require_int(_require_key(ode, "b_grid_points", "models.wear_ode"),
673                                   "models.wear_ode.b_grid_points", 3),
674        b_xtol=b_xtol,
675        b_ftol=b_ftol,
676        ode_solver=solver,
677        rk4_step=_positive_float(ode, "rk4_step", "models.wear_ode"),
678        rk45_rtol=_positive_float(ode, "rk45_rtol", "models.wear_ode"),
679        rk45_atol=_positive_float(ode, "rk45_atol", "models.wear_ode"),
680        candidate_k=candidate_k,
681        candidate_weights=weights,
682        default_k=default_k,
683        default_weights=default_weights,
684        negative_degradation_tolerance_s=_require_float(
685            _require_key(models, "negative_degradation_tolerance_s", "models"),
686            "models.negative_degradation_tolerance_s", 0.0,
687        ),
688        bootstrap_resamples=_require_int(_require_key(models, "bootstrap_resamples", "models"),
689                                         "models.bootstrap_resamples", 2),

953    """Read ``config.yaml`` and return a validated :class:`Config`.

956    ----------
957    path : str or pathlib.Path, optional
958        Path to the YAML configuration file. Default ``"config.yaml"``

963    Config

965
966    Raises
967    ------
968    FileNotFoundError
969        If ``path`` does not exist.
970    ValueError
971        If the file is not valid YAML, or a value is missing/out of range.
972    TypeError
973        If a value has the wrong type.
974
975    Example
976    -------
977    >>> config = load_config("config.yaml")
978    >>> [race.year for race in config.test_races]
979    [2025]
980    """
981    # The file must exist; then parse it as YAML.
982    path = Path(path)
983    if not path.is_file():
984        raise FileNotFoundError(f"Configuration file not found: {path.resolve()}")

989            raw = yaml.safe_load(file)
990        except yaml.YAMLError as error:
991            raise ValueError(f"{path} is not valid YAML: {error}") from error
992    return config_from_dict(raw)
