tyredeg.models.polynomial

Polynomial tyre-degradation model.

lap_time = offset_of_its_stint + d(tyre_age),
d(n) = c_1 n + c_2 n^2 + ... + c_p n^p          (so d(0) = 0)

The degradation coefficients are shared by all stints of one compound; each stint has its own offset (base pace, DECISIONS.md D39). The fit uses my least-squares module (normal equations + my Gaussian elimination, or QR) on the within-stint transformed data (Frisch-Waugh-Lovell, see tyredeg.models.common). Inputs are scaled to [-1, 1] before building the powers (D35) and the coefficients are converted back to tyre-age units.

Candidate degrees are 1, 2, 3; Phase 5 cross-validation picks one (D40).

 44class PolynomialDegradationModel:
 45    """Polynomial degradation d(n) = sum_{k=1..p} c_k n^k with per-stint offsets.
 46
 47    Parameters
 48    ----------
 49    compound : str
 50        Pirelli compound code, e.g. "C3".
 51    degree : int, optional
 52        Polynomial degree p >= 1. Default 1 (linear wear).
 53    scale : bool, optional
 54        Scale tyre age to [-1, 1] before forming powers (D35). Default True.
 55    method : str, optional
 56        "normal" (normal equations + my Gaussian elimination, default) or "qr".
 57    negative_tolerance_s : float, optional
 58        Degradation below -tolerance [s] is flagged (D43). Default 0.01.
 59
 60    Raises
 61    ------
 62    TypeError, ValueError
 63        For an unknown compound, degree < 1, unknown method or negative tolerance.
 64
 65    Example
 66    -------
 67    >>> model = PolynomialDegradationModel("C3", degree=1)
 68    >>> model = model.fit([1, 2, 3, 1, 2, 3], [80.1, 80.2, 80.3, 81.1, 81.2, 81.3],
 69    ...                   stint_id=["a", "a", "a", "b", "b", "b"])
 70    >>> round(float(model.params()["coefficients"][0]), 6)   # c_1 [s/lap]
 71    0.1
 72    """

 74    def __init__(self, compound, degree=DEFAULT_DEGREE, scale=True, method="normal",
 75                 negative_tolerance_s=DEFAULT_NEGATIVE_TOLERANCE_S):
 76        self.compound = check_compound(compound)
 77        if isinstance(degree, bool) or not isinstance(degree, (int, np.integer)) or degree < 1:
 78            raise ValueError(f"degree must be an integer >= 1, got {degree!r}.")
 79        if not isinstance(scale, bool):
 80            raise TypeError("scale must be True or False.")
 81        if method not in VALID_METHODS:
 82            raise ValueError(f"method must be one of {VALID_METHODS}, got {method!r}.")
 83        if not negative_tolerance_s >= 0:
 84            raise ValueError("negative_tolerance_s must be >= 0.")
 85        self.degree = int(degree)
 86        self.scale = scale
 87        self.method = method
 88        self.negative_tolerance_s = float(negative_tolerance_s)
 89        self._fitted = False
 90        self.bootstrap_result = None

 94    # ------------------------------------------------------------------
 95    def _regressors(self, tyre_age):
 96        """Columns s^k - s0^k (k = 1..p), where s0 is the scaled age 0, so d(0) = 0."""
 97        powers = polynomial_design_matrix(tyre_age, self.degree, self._centre, self._half_range)
 98        at_zero = polynomial_design_matrix(np.zeros(1), self.degree, self._centre, self._half_range)
 99        # Column 0 (the constant) is dropped: the stint offsets play that role.
100        return powers[:, 1:] - at_zero[:, 1:]

102    def fit(self, tyre_age, lap_time, stint_id=None):
103        """Fit the degradation coefficients and the stint offsets.
104
105        Parameters
106        ----------
107        tyre_age : array_like
108            Tyre age [laps] of each lap, >= 0.
109        lap_time : array_like
110            Fuel-corrected lap time [s] of each lap, > 0.
111        stint_id : array_like or None, optional
112            Stint identifier per lap; None = one stint.
113
114        Returns
115        -------
116        PolynomialDegradationModel
117            ``self``, fitted.
118
119        Raises
120        ------
121        ValueError
122            For invalid inputs, too few laps (no residual degrees of freedom)
123            or fewer than ``degree + 1`` distinct tyre ages.
124        SingularMatrixError
125            If the regressors are linearly dependent after removing the
126            stint means (e.g. every stint has a single tyre age).
127
128        Example
129        -------
130        >>> PolynomialDegradationModel("C3").fit([1, 2, 3], [80.0, 80.1, 80.2]).params()["n_laps"]
131        3
132        """
133        data = prepare_stint_data(tyre_age, lap_time, stint_id)
134        dof = check_degrees_of_freedom(data, self.degree)
135        if np.unique(data.tyre_age).size < self.degree + 1:
136            raise ValueError(f"A degree-{self.degree} model needs at least "
137                             f"{self.degree + 1} distinct tyre ages.")
138        low, high = float(data.tyre_age.min()), float(data.tyre_age.max())
139        self._centre = (low + high) / 2.0 if self.scale else 0.0
140        self._half_range = (high - low) / 2.0 if self.scale else 1.0
141
142        regressors = self._regressors(data.tyre_age)
143        within_x = np.column_stack([data.demean(column) for column in regressors.T])
144        within_y = data.demean(data.lap_time)
145        if self.method == "normal":
146            result = fit_linear_least_squares(within_x, within_y)
147        else:
148            result = fit_linear_least_squares_qr(within_x, within_y)
149
150        self._data = data
151        self._dof = dof
152        self._scaled_coefficients = result.coefficients
153        self._rss = result.residual_sum_of_squares
154        self._condition = result.condition_estimate
155        self._offsets = data.stint_means(data.lap_time - regressors @ result.coefficients)
156
157        # Covariance of the scaled coefficients: sigma^2 (Zw^T Zw)^-1, with the
158        # residual degrees of freedom corrected for the stint offsets.
159        sigma_squared = self._rss / dof
160        lu, permutation = lu_factor(within_x.T @ within_x)
161        scaled_cov = sigma_squared * lu_solve(lu, permutation, np.eye(self.degree))
162        # Convert to tyre-age units. Row/column 0 of T is the constant term,
163        # which is identically 0 for d, so it is dropped.
164        transform = unscaling_matrix(self.degree, self._centre, self._half_range)[1:, 1:]
165        self._coefficients = transform @ self._scaled_coefficients
166        self._covariance = transform @ scaled_cov @ transform.T
167        self._fitted = True
168        self.bootstrap_result = None
169        self._constraint = check_non_negative(self.degradation, high, self.negative_tolerance_s)
170        return self
171
172    def _require_fitted(self):
173        if not self._fitted:
174            raise NotFittedError("Call fit() before using the model.")

179    def degradation(self, tyre_age):
180        """Degradation d(n) [s] relative to a new tyre (d(0) = 0).
181
182        Parameters
183        ----------
184        tyre_age : float or array_like
185            Tyre age(s) [laps], >= 0.
186
187        Returns
188        -------
189        numpy.ndarray
190
191        Raises
192        ------
193        NotFittedError
194            Before ``fit``.
195        ValueError
196            For invalid ages.
197
198        Example
199        -------
200        >>> model = PolynomialDegradationModel("C3").fit([1, 2, 3], [80.0, 80.1, 80.2])
201        >>> model.degradation([0.0, 10.0]).round(6).tolist()
202        [0.0, 1.0]
203        """
204        self._require_fitted()
205        age = check_tyre_age(tyre_age)
206        return self._regressors(age) @ self._scaled_coefficients

208    def predict(self, tyre_age, base_time=None):
209        """Predicted lap time [s] = base_time + d(tyre_age).
210
211        Parameters
212        ----------
213        tyre_age : float or array_like
214            Tyre age(s) [laps], >= 0.
215        base_time : float or None, optional
216            Lap time of a new tyre [s]. Default: the median stint offset of
217            the training data (a "typical" car).
218
219        Returns
220        -------
221        numpy.ndarray
222            Predicted fuel-corrected lap times [s], one per age.
223
224        Raises
225        ------
226        NotFittedError
227            Before ``fit``.
228        ValueError
229            For invalid ages or a non-positive ``base_time``.
230
231        Example
232        -------
233        >>> model = PolynomialDegradationModel("C3").fit([1, 2, 3], [80.0, 80.1, 80.2])
234        >>> model.predict([4.0], base_time=90.0).round(6).tolist()
235        [90.4]
236        """
237        self._require_fitted()
238        base = self.base_time if base_time is None else float(base_time)
239        if not np.isfinite(base) or base <= 0:
240            raise ValueError(f"base_time must be a positive number of seconds, got {base_time!r}.")
241        return base + self.degradation(tyre_age)

273    def parameter_names(self):
274        """Names of the degradation coefficients, e.g. ('c1 [s/lap]', 'c2 [s/lap^2]').
275
276        Returns
277        -------
278        tuple of str
279            One name (with unit) per coefficient c1 .. c_degree.

282        ------

284

288        ('c1 [s/lap]', 'c2 [s/lap^2]')

295    def bootstrap(self, n_resamples, confidence_level=0.95, seed=42):
296        """Stint-bootstrap uncertainty of the coefficients (D45); stored for summary().
297
298        Parameters
299        ----------
300        n_resamples : int
301            Number of resamples >= 2 (config: 200).
302        confidence_level : float, optional
303            In (0, 1). Default 0.95.
304        seed : int, optional
305            Random seed (rule 11). Default 42.
306
307        Returns
308        -------
309        BootstrapResult
310
311        Raises
312        ------
313        NotFittedError
314            Before ``fit``.
315        ValueError
316            For an invalid input (see Parameters).
317
318        Example
319        -------
320        >>> result = model.bootstrap(200, seed=42)  # doctest: +SKIP
321        """
322        self._require_fitted()
323
324        def refit(ages, times, ids):
325            clone = PolynomialDegradationModel(self.compound, self.degree, self.scale,
326                                               self.method, self.negative_tolerance_s)
327            return clone.fit(ages, times, ids)._coefficients
328
329        self.bootstrap_result = stint_bootstrap(self._data, refit, self.parameter_names(),
330                                                n_resamples, confidence_level, seed)
331        return self.bootstrap_result

333    def params(self):
334        """Fitted parameters and diagnostics as a dictionary.
335
336        Returns
337        -------
338        dict
339            Keys: compound, model, degree, coefficients (c_1..c_p in tyre-age
340            units), standard_errors (analytic), bootstrap_standard_errors
341            (None until :meth:`bootstrap`), base_time, n_laps, n_stints, rss,
342            residual_std, condition_estimate, min_degradation,
343            negative_degradation_flag.
344
345        Raises
346        ------
347        NotFittedError
348            Before ``fit``.
349
350        Example
351        -------
352        >>> PolynomialDegradationModel("C3").fit([1, 2, 3], [80.0, 80.1, 80.2]).params()["degree"]
353        1
354        """
355        self._require_fitted()
356        return {
357            "compound": self.compound,
358            "model": "polynomial",
359            "degree": self.degree,
360            "coefficients": self._coefficients.copy(),
361            "standard_errors": np.sqrt(np.diag(self._covariance)),
362            "bootstrap_standard_errors": (None if self.bootstrap_result is None
363                                          else self.bootstrap_result.standard_errors.copy()),
364            "base_time": self.base_time,
365            "n_laps": self._data.n_laps,
366            "n_stints": self._data.n_stints,
367            "rss": self._rss,
368            "residual_std": float(np.sqrt(self._rss / self._dof)),
369            "condition_estimate": self._condition,
370            "min_degradation": self._constraint.min_degradation_s,
371            "negative_degradation_flag": self._constraint.violated,
372        }

374    def summary(self):
375        """Human-readable report of the fit (str).
376
377        Returns
378        -------
379        str
380            Lines with the data size, residual standard deviation, base time, each
381            coefficient with its standard errors, and the non-negative check.
382
383        Raises
384        ------
385        NotFittedError
386            Before ``fit``.
387
388        Example
389        -------
390        >>> ages = np.tile(np.arange(1.0, 11.0), 2)
391        >>> model = PolynomialDegradationModel("C3").fit(ages, 80.0 + 0.06 * ages + 0.01 * np.cos(ages), [0] * 10 + [1] * 10)
392        >>> print(model.summary().splitlines()[0])
393        Polynomial degradation model, compound C3, degree 1
394        """
395        p = self.params()
396        lines = [
397            f"Polynomial degradation model, compound {p['compound']}, degree {p['degree']}",
398            f"  {p['n_laps']} laps in {p['n_stints']} stints; residual std {p['residual_std']:.3f} s; "
399            f"base time (median offset) {p['base_time']:.3f} s",
400        ]
401        lines += format_parameter_table(self.parameter_names(), p["coefficients"],
402                                        p["standard_errors"], self.bootstrap_result)
403        status = "VIOLATED" if p["negative_degradation_flag"] else "ok"
404        lines.append(f"  non-negative degradation: {status} (min d = {p['min_degradation']:.4f} s)")
405        return "\n".join(lines)

103    def fit(self, tyre_age, lap_time, stint_id=None):
104        """Fit the degradation coefficients and the stint offsets.
105
106        Parameters
107        ----------
108        tyre_age : array_like
109            Tyre age [laps] of each lap, >= 0.
110        lap_time : array_like
111            Fuel-corrected lap time [s] of each lap, > 0.
112        stint_id : array_like or None, optional
113            Stint identifier per lap; None = one stint.
114
115        Returns
116        -------
117        PolynomialDegradationModel
118            ``self``, fitted.
119
120        Raises
121        ------
122        ValueError
123            For invalid inputs, too few laps (no residual degrees of freedom)
124            or fewer than ``degree + 1`` distinct tyre ages.
125        SingularMatrixError
126            If the regressors are linearly dependent after removing the
127            stint means (e.g. every stint has a single tyre age).
128
129        Example
130        -------
131        >>> PolynomialDegradationModel("C3").fit([1, 2, 3], [80.0, 80.1, 80.2]).params()["n_laps"]
132        3
133        """
134        data = prepare_stint_data(tyre_age, lap_time, stint_id)
135        dof = check_degrees_of_freedom(data, self.degree)
136        if np.unique(data.tyre_age).size < self.degree + 1:
137            raise ValueError(f"A degree-{self.degree} model needs at least "
138                             f"{self.degree + 1} distinct tyre ages.")
139        low, high = float(data.tyre_age.min()), float(data.tyre_age.max())
140        self._centre = (low + high) / 2.0 if self.scale else 0.0
141        self._half_range = (high - low) / 2.0 if self.scale else 1.0
142
143        regressors = self._regressors(data.tyre_age)
144        within_x = np.column_stack([data.demean(column) for column in regressors.T])
145        within_y = data.demean(data.lap_time)
146        if self.method == "normal":
147            result = fit_linear_least_squares(within_x, within_y)
148        else:
149            result = fit_linear_least_squares_qr(within_x, within_y)
150
151        self._data = data
152        self._dof = dof
153        self._scaled_coefficients = result.coefficients
154        self._rss = result.residual_sum_of_squares
155        self._condition = result.condition_estimate
156        self._offsets = data.stint_means(data.lap_time - regressors @ result.coefficients)
157
158        # Covariance of the scaled coefficients: sigma^2 (Zw^T Zw)^-1, with the
159        # residual degrees of freedom corrected for the stint offsets.
160        sigma_squared = self._rss / dof
161        lu, permutation = lu_factor(within_x.T @ within_x)
162        scaled_cov = sigma_squared * lu_solve(lu, permutation, np.eye(self.degree))
163        # Convert to tyre-age units. Row/column 0 of T is the constant term,
164        # which is identically 0 for d, so it is dropped.
165        transform = unscaling_matrix(self.degree, self._centre, self._half_range)[1:, 1:]
166        self._coefficients = transform @ self._scaled_coefficients
167        self._covariance = transform @ scaled_cov @ transform.T
168        self._fitted = True
169        self.bootstrap_result = None
170        self._constraint = check_non_negative(self.degradation, high, self.negative_tolerance_s)
171        return self
